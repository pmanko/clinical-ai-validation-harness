SELECT date_trunc('MONTH', obs_date) AS month,
       code_display AS test_name,
       count(*) AS result_count,
       percentile_approx(val_quantity, 0.5) AS median_value
FROM observation_flat
WHERE obs_date >= :since_date
GROUP BY 1, 2
ORDER BY 1
